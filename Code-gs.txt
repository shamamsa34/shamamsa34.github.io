/**
 * موقع شمامسة تالتة ورابعة — خدمة الشمامسة
 * الشيت ده هو قاعدة البيانات:
 *   Kids        أسماء الأطفال
 *   Lessons     الدروس (القسم، الاسم، لينك الفيديو أو الفويس)
 *   Questions   أسئلة كل درس
 *   Results     نتيجة كل طفل في كل درس (والألحان اللي سمّعها)
 *   Answers     تفاصيل كل سؤال: الطفل اختار إيه والصح كان إيه
 *   Bonus       نقط الحصة اللي الخادم بيضيفها (حضور القداس، الحصة، …)
 *   Recordings  تسجيلات تسميع الألحان (الملفات نفسها في فولدر على Drive)
 *
 * بعد أي تحديث للكود: شغّل setup مرة واحدة (Run) ووافق على الصلاحيات،
 * وبعدين Deploy > Manage deployments > ✏️ > New version > Deploy.
 */

const APP_TITLE = 'شمامسة تالتة ورابعة';
const RECITE_SECTION = 'الحان';
const MAX_MEDIA_BYTES = 18 * 1024 * 1024;   // أكبر فويس/فيديو Drive يتشغّل جوه الموقع
const MAX_REC_BYTES = 12 * 1024 * 1024;     // أكبر تسجيل تسميع

const DEFAULT_CATS = [
  { name: 'حضور القداس', points: 20 },
  { name: 'حضور الحصة', points: 10 },
  { name: 'تسميع لحن الأسبوع', points: 15 },
  { name: 'man of the day', points: 10 },
  { name: 'تفاعل في الحصة', points: 1 }
];
const POINT_DEFAULTS = { solve: 5, correct: 0, full: 0, recite: 15 };

const HEAD = {
  Kids:       ['id', 'name', 'class', 'active'],
  Lessons:    ['id', 'section', 'title', 'videoUrl', 'description', 'order', 'active', 'updatedAt'],
  Questions:  ['lessonId', 'order', 'question', 'option1', 'option2', 'option3', 'option4', 'correct', 'videoUrl'],
  Results:    ['timestamp', 'kidId', 'kidName', 'class', 'section', 'lessonId', 'lessonTitle', 'score', 'total'],
  Answers:    ['timestamp', 'kidId', 'kidName', 'class', 'section', 'lessonTitle', 'questionNo', 'question', 'chosen', 'correctAnswer', 'isCorrect'],
  Bonus:      ['id', 'timestamp', 'date', 'kidId', 'kidName', 'class', 'reason', 'points'],
  Recordings: ['id', 'timestamp', 'kidId', 'kidName', 'class', 'lessonId', 'lessonTitle', 'seconds', 'heard', 'fileUrl']
};

/* ---------- web app ---------- */
// Without parameters the link shows the site itself. The site on GitHub talks to doPost (JSON).
function doGet(e) {
  if (e && e.parameter && e.parameter.fn) {
    let args = [];
    try { args = JSON.parse(e.parameter.args || '[]'); } catch (x) {}
    return apiOut_(e.parameter.fn, args);
  }
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle(APP_TITLE)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function doPost(e) {
  let body = {};
  try { body = JSON.parse(e.postData.contents); } catch (x) {}
  return apiOut_(body.fn, body.args || []);
}

function apiOut_(fn, args) {
  const API = {
    getPublicData: getPublicData, getKidProgress: getKidProgress, submitResult: submitResult,
    getLeaderboard: getLeaderboard, getMedia: getMedia, saveRecording: saveRecording,
    adminGetData: adminGetData, adminSaveLesson: adminSaveLesson, adminDeleteLesson: adminDeleteLesson,
    adminAddKids: adminAddKids, adminRemoveKid: adminRemoveKid, adminUpdateKid: adminUpdateKid, adminSavePoints: adminSavePoints,
    adminChangePassword: adminChangePassword, adminAddBonus: adminAddBonus, adminDeleteBonus: adminDeleteBonus,
    adminSaveCategories: adminSaveCategories, adminGetRecording: adminGetRecording,
    adminMarkRecording: adminMarkRecording, adminUploadMedia: adminUploadMedia
  };
  let out;
  try {
    if (!Object.prototype.hasOwnProperty.call(API, fn)) throw new Error('طلب مش معروف');
    out = { ok: true, data: API[fn].apply(null, Array.isArray(args) ? args : []) };
  } catch (err) {
    out = { ok: false, error: (err && err.message) || String(err) };
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

/* ---------- sheet helpers ---------- */
function ss_() {
  const active = SpreadsheetApp.getActiveSpreadsheet();
  if (active) return active;
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('SHEET_ID');
  if (id) { try { return SpreadsheetApp.openById(id); } catch (e) {} }
  const ss = SpreadsheetApp.create('موقع شمامسة تالتة ورابعة - البيانات');
  props.setProperty('SHEET_ID', ss.getId());
  return ss;
}

function sh_(name) {
  const ss = ss_();
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.getRange(1, 1, 1, HEAD[name].length).setValues([HEAD[name]]).setFontWeight('bold').setBackground('#1f4aa8').setFontColor('#ffffff');
    sh.setFrozenRows(1);
    sh.setRightToLeft(true);
  } else if (sh.getLastColumn() < HEAD[name].length) {
    sh.getRange(1, 1, 1, HEAD[name].length).setValues([HEAD[name]]).setFontWeight('bold').setBackground('#1f4aa8').setFontColor('#ffffff');
  }
  return sh;
}

function read_(name) {
  const sh = sh_(name), h = HEAD[name], last = sh.getLastRow();
  if (last < 2) return [];
  const tz = ss_().getSpreadsheetTimeZone();
  return sh.getRange(2, 1, last - 1, h.length).getValues()
    .filter(r => r.some(c => c !== '' && c !== null))
    .map(r => {
      const o = {};
      h.forEach((k, i) => {
        const v = r[i];
        o[k] = v instanceof Date ? (k === 'date' ? Utilities.formatDate(v, tz, 'yyyy-MM-dd') : v.toISOString()) : v;
      });
      return o;
    });
}

function writeAll_(name, objs) {
  const sh = sh_(name), h = HEAD[name], last = sh.getLastRow();
  if (last > 1) sh.getRange(2, 1, last - 1, h.length).clearContent();
  if (objs.length) sh.getRange(2, 1, objs.length, h.length).setValues(objs.map(o => h.map(k => o[k] === undefined ? '' : o[k])));
}

function append_(name, objs) {
  if (!objs.length) return;
  const sh = sh_(name), h = HEAD[name];
  sh.getRange(sh.getLastRow() + 1, 1, objs.length, h.length).setValues(objs.map(o => h.map(k => o[k] === undefined ? '' : o[k])));
}

function truthy_(v) {
  if (v === '' || v === null || v === undefined) return true;
  return v === true || v === 1 || ['true', '1', 'نعم', 'yes'].indexOf(String(v).trim().toLowerCase()) >= 0;
}
function newId_(p) { return p + Utilities.getUuid().replace(/-/g, '').slice(0, 8); }
function today_() { return Utilities.formatDate(new Date(), ss_().getSpreadsheetTimeZone(), 'yyyy-MM-dd'); }

function withLock_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try { return fn(); } finally { lock.releaseLock(); }
}

function driveId_(url) {
  const m = String(url || '').match(/drive\.google\.com\/(?:file\/d\/|open\?id=|uc\?(?:.*&)?id=)([\w-]+)/);
  return m ? m[1] : '';
}

function folder_(prop, name) {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty(prop);
  if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} }
  const f = DriveApp.createFolder(name);
  props.setProperty(prop, f.getId());
  return f;
}

function kids_() {
  const kids = read_('Kids');
  let changed = false;
  kids.forEach(k => {
    if (!k.id && k.name) { k.id = newId_('k'); changed = true; }
    k.id = String(k.id); k.name = String(k.name).trim(); k.class = String(k.class || '').trim();
    k.active = truthy_(k.active);
  });
  if (changed) withLock_(() => writeAll_('Kids', kids));
  return kids.filter(k => k.name);
}

function lessons_(includeHidden) {
  const qs = read_('Questions');
  return read_('Lessons')
    .filter(l => l.id && (includeHidden || truthy_(l.active)))
    .map(l => {
      const section = String(l.section).trim();
      const questions = section === RECITE_SECTION ? [] : qs.filter(q => String(q.lessonId) === String(l.id))
        .sort((a, b) => Number(a.order) - Number(b.order))
        .map(q => {
          const raw = [q.option1, q.option2, q.option3, q.option4].map(x => String(x === null || x === undefined ? '' : x).trim());
          const correctN = Number(q.correct) || 1;
          const options = []; let correct = -1;
          raw.forEach((t, i) => { if (t) { if (i + 1 === correctN) correct = options.length; options.push(t); } });
          return { question: String(q.question), options: options, correct: correct, raw: raw, correctN: correctN, videoUrl: String(q.videoUrl || '').trim() };
        })
        .filter(q => q.question && q.options.length >= 2 && q.correct >= 0);
      return {
        id: String(l.id), section: section, title: String(l.title),
        videoUrl: String(l.videoUrl || ''), description: String(l.description || ''),
        order: Number(l.order) || 0, active: truthy_(l.active), recite: section === RECITE_SECTION, questions: questions
      };
    })
    .sort((a, b) => a.order - b.order);
}

/* ---------- points & categories (set from the admin dashboard) ---------- */
function points_() {
  const p = PropertiesService.getScriptProperties();
  const n = (k, d) => { const v = p.getProperty('POINTS_' + k.toUpperCase()); return v === null || v === '' ? d : Number(v) || 0; };
  return { solve: n('solve', POINT_DEFAULTS.solve), correct: n('correct', POINT_DEFAULTS.correct), full: n('full', POINT_DEFAULTS.full), recite: n('recite', POINT_DEFAULTS.recite) };
}
function categories_() {
  const v = PropertiesService.getScriptProperties().getProperty('BONUS_CATS');
  if (v) { try { const a = JSON.parse(v); if (Array.isArray(a)) return a; } catch (e) {} }
  return DEFAULT_CATS;
}
function lessonPoints_(rules, section, score, total) {
  if (section === RECITE_SECTION) return rules.recite;
  return rules.solve + rules.correct * score + (total && score === total ? rules.full : 0);
}

/* ---------- kids' side ---------- */
function getPublicData() {
  return {
    kids: kids_().filter(k => k.active).map(k => ({ id: k.id, name: k.name, class: k.class })),
    lessons: lessons_(false),
    rules: points_()
  };
}

function getKidProgress(kidId) {
  const out = {};
  read_('Results').filter(r => String(r.kidId) === String(kidId)).forEach(r => {
    const s = Number(r.score) || 0, b = out[r.lessonId];
    if (!b || s > b.score) out[r.lessonId] = { score: s, total: Number(r.total) || 0 };
  });
  return out;
}

function submitResult(p) {
  const kid = kids_().find(k => k.id === String(p.kidId));
  if (!kid) throw new Error('الاسم ده مش موجود في القايمة');
  const lesson = lessons_(true).find(l => l.id === String(p.lessonId));
  if (!lesson) throw new Error('الدرس ده مش موجود');
  const now = new Date();
  let score = 0;
  const answers = lesson.questions.map((q, i) => {
    const a = (p.answers || []).find(x => Number(x.q) === i);
    const chosen = a ? Number(a.chosen) : -1;
    const ok = chosen === q.correct;
    if (ok) score++;
    return {
      timestamp: now, kidId: kid.id, kidName: kid.name, class: kid.class, section: lesson.section,
      lessonTitle: lesson.title, questionNo: i + 1, question: q.question,
      chosen: chosen >= 0 && chosen < q.options.length ? q.options[chosen] : '',
      correctAnswer: q.options[q.correct], isCorrect: ok ? 'صح' : 'غلط'
    };
  });
  withLock_(() => {
    append_('Answers', answers);
    append_('Results', [{
      timestamp: now, kidId: kid.id, kidName: kid.name, class: kid.class, section: lesson.section,
      lessonId: lesson.id, lessonTitle: lesson.title, score: score, total: lesson.questions.length
    }]);
  });
  return { score: score, total: lesson.questions.length };
}

/* Drive audio/video for a lesson (q = -1) or one of its questions, sent to the page so it plays inside it */
function getMedia(lessonId, q) {
  const l = lessons_(true).find(x => x.id === String(lessonId));
  if (!l) throw new Error('الدرس ده مش موجود');
  const qi = Number(q);
  const url = (q === undefined || q === null || qi < 0) ? l.videoUrl : (l.questions[qi] || {}).videoUrl;
  const id = driveId_(url);
  if (!id) throw new Error('اللينك ده مش ملف Drive');
  const f = DriveApp.getFileById(id);
  if (f.getSize() > MAX_MEDIA_BYTES) return { tooBig: true };
  const b = f.getBlob();
  return { mime: b.getContentType(), name: f.getName(), data: Utilities.base64Encode(b.getBytes()) };
}

/* A kid's recitation: saved as an audio file in Drive + a row in Recordings + a result for the points */
function saveRecording(p) {
  const kid = kids_().find(k => k.id === String(p.kidId));
  if (!kid) throw new Error('الاسم ده مش موجود في القايمة');
  const lesson = lessons_(true).find(l => l.id === String(p.lessonId));
  if (!lesson || !lesson.recite) throw new Error('اللحن ده مش موجود');
  const bytes = Utilities.base64Decode(String(p.data || ''));
  if (!bytes.length) throw new Error('التسجيل فاضي، سجّل تاني');
  if (bytes.length > MAX_REC_BYTES) throw new Error('التسجيل طويل أوي، سجّل أقصر شوية');
  const mime = String(p.mime || 'audio/webm').split(';')[0];
  const ext = /mp4|m4a|aac/.test(mime) ? 'm4a' : /ogg/.test(mime) ? 'ogg' : /wav/.test(mime) ? 'wav' : 'webm';
  const now = new Date();
  const stamp = Utilities.formatDate(now, ss_().getSpreadsheetTimeZone(), 'yyyy-MM-dd HH-mm');
  const file = folder_('REC_FOLDER', 'تسميع الأطفال - شمامسة تالتة ورابعة')
    .createFile(Utilities.newBlob(bytes, mime, kid.name + ' - ' + lesson.title + ' - ' + stamp + '.' + ext));
  withLock_(() => {
    append_('Recordings', [{
      id: file.getId(), timestamp: now, kidId: kid.id, kidName: kid.name, class: kid.class,
      lessonId: lesson.id, lessonTitle: lesson.title, seconds: Number(p.seconds) || 0, heard: false, fileUrl: file.getUrl()
    }]);
    append_('Results', [{
      timestamp: now, kidId: kid.id, kidName: kid.name, class: kid.class, section: lesson.section,
      lessonId: lesson.id, lessonTitle: lesson.title, score: 1, total: 1
    }]);
  });
  return { points: points_().recite };
}

/* Each lesson counts once per kid (best attempt) + all session points the servants added */
function getLeaderboard() {
  const rules = points_();
  const t = {};
  kids_().filter(k => k.active).forEach(k => {
    t[k.id] = { kidId: k.id, name: k.name, class: k.class, points: 0, lessons: 0, bonus: 0, bySection: {} };
  });
  const best = {};
  read_('Results').forEach(r => {
    const key = r.kidId + '|' + r.lessonId, s = Number(r.score) || 0;
    if (!best[key] || s > best[key].score) best[key] = { kidId: String(r.kidId), section: String(r.section), score: s, total: Number(r.total) || 0 };
  });
  Object.keys(best).forEach(key => {
    const b = best[key], o = t[b.kidId];
    if (!o) return;
    const pts = lessonPoints_(rules, b.section, b.score, b.total);
    o.points += pts; o.lessons++;
    o.bySection[b.section] = (o.bySection[b.section] || 0) + pts;
  });
  read_('Bonus').forEach(x => {
    const o = t[String(x.kidId)];
    if (!o) return;
    const p = Number(x.points) || 0;
    o.points += p; o.bonus += p;
    o.bySection['حصة'] = (o.bySection['حصة'] || 0) + p;
  });
  return { rows: Object.keys(t).map(k => t[k]), rules: rules, cats: categories_() };
}

/* ---------- admin ---------- */
function password_() {
  return PropertiesService.getScriptProperties().getProperty('ADMIN_PASSWORD') || '1234';
}
function auth_(pw) {
  if (String(pw) !== password_()) throw new Error('كلمة السر غلط');
}

function adminGetData(pw) {
  auth_(pw);
  const ss = ss_();
  return {
    kids: kids_(),
    lessons: lessons_(true),
    results: read_('Results').slice(-1500).reverse(),
    bonus: read_('Bonus').slice(-2000).reverse(),
    recordings: read_('Recordings').slice(-800).reverse(),
    rules: points_(),
    cats: categories_(),
    today: today_(),
    sheetUrl: ss.getUrl(),
    xlsxUrl: 'https://docs.google.com/spreadsheets/d/' + ss.getId() + '/export?format=xlsx'
  };
}

function adminSaveLesson(pw, l) {
  auth_(pw);
  if (!l || !String(l.title || '').trim()) throw new Error('اكتب اسم الدرس');
  const recite = String(l.section) === RECITE_SECTION;
  if (!recite && (!l.questions || !l.questions.length)) throw new Error('الدرس محتاج سؤال واحد على الأقل');
  withLock_(() => {
    const lessons = read_('Lessons');
    const id = l.id ? String(l.id) : newId_('L');
    const i = lessons.findIndex(x => String(x.id) === id);
    const maxOrder = lessons.reduce((m, x) => Math.max(m, Number(x.order) || 0), 0);
    const row = {
      id: id, section: String(l.section), title: String(l.title).trim(),
      videoUrl: String(l.videoUrl || '').trim(), description: String(l.description || '').trim(),
      order: i >= 0 ? lessons[i].order : maxOrder + 1, active: !!l.active, updatedAt: new Date()
    };
    if (i >= 0) lessons[i] = row; else lessons.push(row);
    writeAll_('Lessons', lessons);

    const others = read_('Questions').filter(q => String(q.lessonId) !== id);
    const mine = recite ? [] : l.questions.map((q, n) => {
      const raw = (q.raw || []).concat(['', '', '', '']).slice(0, 4);
      return { lessonId: id, order: n + 1, question: String(q.question).trim(),
        option1: raw[0], option2: raw[1], option3: raw[2], option4: raw[3], correct: Number(q.correctN) || 1,
        videoUrl: String(q.videoUrl || '').trim() };
    });
    writeAll_('Questions', others.concat(mine));
  });
  return adminGetData(pw);
}

function adminDeleteLesson(pw, id) {
  auth_(pw);
  withLock_(() => {
    writeAll_('Lessons', read_('Lessons').filter(x => String(x.id) !== String(id)));
    writeAll_('Questions', read_('Questions').filter(q => String(q.lessonId) !== String(id)));
  });
  return adminGetData(pw);
}

function adminAddKids(pw, names, cls) {
  auth_(pw);
  const existing = {};
  read_('Kids').forEach(k => { if (truthy_(k.active)) existing[String(k.name).trim()] = true; });
  const rows = (names || []).map(n => String(n).trim()).filter(n => n && !existing[n])
    .map(n => ({ id: newId_('k'), name: n, class: String(cls || ''), active: true }));
  withLock_(() => append_('Kids', rows));
  return adminGetData(pw);
}

/* Fix a kid's name or class. Points stay with the kid (they're linked by id, not by name). */
function adminUpdateKid(pw, id, name, cls) {
  auth_(pw);
  const n = String(name || '').trim();
  if (!n) throw new Error('اكتب الاسم');
  withLock_(() => {
    const kids = read_('Kids');
    const k = kids.find(x => String(x.id) === String(id));
    if (!k) throw new Error('الاسم ده مش موجود');
    k.name = n; if (cls) k.class = String(cls);
    writeAll_('Kids', kids);
  });
  return adminGetData(pw);
}

function adminRemoveKid(pw, id) {
  auth_(pw);
  withLock_(() => {
    const kids = read_('Kids');
    kids.forEach(k => { if (String(k.id) === String(id)) k.active = false; });
    writeAll_('Kids', kids);
  });
  return adminGetData(pw);
}

function adminSavePoints(pw, r) {
  auth_(pw);
  const p = PropertiesService.getScriptProperties();
  ['solve', 'correct', 'full', 'recite'].forEach(k => {
    const v = Math.max(0, Math.round(Number(r && r[k]) || 0));
    p.setProperty('POINTS_' + k.toUpperCase(), String(v));
  });
  return adminGetData(pw);
}

function adminChangePassword(pw, newPw) {
  auth_(pw);
  if (String(newPw || '').length < 4) throw new Error('كلمة السر لازم تبقى 4 حروف أو أرقام على الأقل');
  PropertiesService.getScriptProperties().setProperty('ADMIN_PASSWORD', String(newPw));
  return true;
}

/* Session points: one category for several kids at once (or a quick −1/+1/+5) */
function adminAddBonus(pw, kidIds, reason, points, date) {
  auth_(pw);
  const pts = Math.round(Number(points));
  if (!pts) throw new Error('اكتب عدد النقط');
  const why = String(reason || '').trim() || 'نقط الحصة';
  const day = /^\d{4}-\d{2}-\d{2}$/.test(String(date || '')) ? String(date) : today_();
  const byId = {};
  kids_().forEach(k => { byId[k.id] = k; });
  const now = new Date();
  const rows = (kidIds || []).map(String).filter(id => byId[id]).map(id => ({
    id: newId_('b'), timestamp: now, date: day, kidId: id, kidName: byId[id].name, class: byId[id].class, reason: why, points: pts
  }));
  if (!rows.length) throw new Error('اختار طفل واحد على الأقل');
  withLock_(() => append_('Bonus', rows));
  return adminGetData(pw);
}

function adminDeleteBonus(pw, id) {
  auth_(pw);
  withLock_(() => writeAll_('Bonus', read_('Bonus').filter(x => String(x.id) !== String(id))));
  return adminGetData(pw);
}

function adminSaveCategories(pw, cats) {
  auth_(pw);
  const clean = (cats || []).map(c => ({ name: String(c.name || '').trim(), points: Math.round(Number(c.points) || 0) }))
    .filter(c => c.name && c.points).slice(0, 20);
  PropertiesService.getScriptProperties().setProperty('BONUS_CATS', JSON.stringify(clean));
  return adminGetData(pw);
}

function adminGetRecording(pw, id) {
  auth_(pw);
  if (!read_('Recordings').some(r => String(r.id) === String(id))) throw new Error('التسجيل ده مش موجود');
  const f = DriveApp.getFileById(String(id));
  const b = f.getBlob();
  return { mime: b.getContentType(), data: Utilities.base64Encode(b.getBytes()) };
}

function adminMarkRecording(pw, id, heard) {
  auth_(pw);
  withLock_(() => {
    const rows = read_('Recordings');
    rows.forEach(r => { if (String(r.id) === String(id)) r.heard = !!heard; });
    writeAll_('Recordings', rows);
  });
  return adminGetData(pw);
}

/* A voice or video the servant uploads or records from the dashboard → Drive, shared by link */
function adminUploadMedia(pw, f) {
  auth_(pw);
  const bytes = Utilities.base64Decode(String((f && f.data) || ''));
  if (!bytes.length) throw new Error('الملف فاضي');
  if (bytes.length > MAX_MEDIA_BYTES) throw new Error('الملف أكبر من 18 ميجا. صغّره أو ارفعه على YouTube');
  const mime = String((f && f.mime) || 'audio/webm').split(';')[0];
  const file = folder_('MEDIA_FOLDER', 'ملفات دروس الشمامسة').createFile(Utilities.newBlob(bytes, mime, String(f.name || 'فويس')));
  try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
  return { url: 'https://drive.google.com/file/d/' + file.getId() + '/view' };
}

function importKids() {
  const existing = {};
  read_('Kids').forEach(k => { existing[String(k.name).trim()] = true; });
  const rows = KIDS_LIST.filter(r => !existing[r[0]])
    .map(r => ({ id: newId_('k'), name: r[0], class: r[1], active: true }));
  withLock_(() => append_('Kids', rows));
  Logger.log('اتضاف ' + rows.length + ' اسم');
}
